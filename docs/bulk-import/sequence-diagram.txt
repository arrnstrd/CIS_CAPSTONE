sequenceDiagram
    actor Admin
    participant UI as Bulk Import UI
    participant Controller as BulkImportController
    participant Service as BulkImportService
    participant Parser as SpreadsheetParser
    participant Validator as ImportRowValidator
    participant Resolver as SectionResolver
    participant Processor as ImportProcessor
    participant StudentSvc as StudentService
    participant EnrollSvc as EnrollmentService
    participant DB as Database

    Admin->>UI: Select SF-1 Excel file
    Admin->>UI: Click Upload

    UI->>Controller: POST /bulk-import/upload
    Controller->>Service: upload(file)

    Service->>DB: Create BulkImport (status=pending)
    Service-->>Controller: Import Session
    Controller-->>UI: Upload Successful

    Admin->>UI: Click Validate

    UI->>Controller: POST /bulk-import/{id}/validate
    Controller->>Service: validate(import)

    Service->>Parser: Parse Excel (SF-1)
    Parser-->>Service: Parsed Rows

    loop Every Row
        Service->>Validator: Validate row format

        Validator-->>Service: Validation Result

        alt Valid Row
            Service->>Resolver: Resolve Section
            Resolver-->>Service: Section Found
        else Invalid Row
            Service->>DB: Save BulkImportIssue
        end
    end

    Service->>DB: Update validation summary
    Service-->>Controller: Validation Result
    Controller-->>UI: Show Validation Summary Modal

    alt Admin proceeds
        Admin->>UI: Confirm Import

        UI->>Controller: POST /bulk-import/{id}/confirm
        Controller->>Service: confirm(import)

        Service->>Processor: process(validRows)

        loop Every Valid Row

            Processor->>StudentSvc: createStudent(studentData, guardianData)

            StudentSvc->>DB: Begin Transaction
            StudentSvc->>DB: Insert Student
            StudentSvc->>DB: Generate Student Number
            StudentSvc->>DB: Insert Guardian
            StudentSvc->>DB: Generate QR Code
            StudentSvc->>DB: Commit Transaction

            alt Enrollment Data Present

                Processor->>EnrollSvc: createEnrollment(enrollmentData)

                alt Enrollment Success
                    EnrollSvc->>DB: Insert Enrollment
                else Section Full / Duplicate / Invalid
                    EnrollSvc-->>Processor: RuntimeException
                    Processor->>DB: Save BulkImportIssue
                end

            end

            Processor->>DB: Update Progress

        end

        Processor-->>Service: Processing Summary

        alt No Issues
            Service->>DB: Update Status = Completed
        else With Issues
            Service->>DB: Update Status = Completed With Issues
        end

        Service-->>Controller: Import Result
        Controller-->>UI: Success Modal

    else Admin cancels
        UI->>Controller: Cancel Import
        Controller->>Service: cancel(import)
        Service->>DB: Update Status = Cancelled
        Service-->>Controller: Cancelled
        Controller-->>UI: Show Cancel Confirmation
    end

    Admin->>UI: Open Import History
    UI->>Controller: GET /bulk-import/history
    Controller->>DB: Retrieve History
    DB-->>Controller: Import Logs
    Controller-->>UI: Display History Table

    Admin->>UI: View Import Details
    UI->>Controller: GET /bulk-import/{id}/issues
    Controller->>DB: Retrieve Issues
    DB-->>Controller: Issue List
    Controller-->>UI: Display Issues Page
